// Integration Examples

Works with any stack.

If it can make an HTTP request, it can use AILeash.

Python 
import requests

# One function. Drop into any AI decision point.
def govern(device_id, action, context="", risk_score=0.0):
    r = requests.post(
        "https://sebbi.pro/api/govern",
        headers={"Authorization": f"Bearer {API_KEY}"},
        json={
            "device_id": device_id,
            "action": action,
            "context": context,
            "risk_score": risk_score
        }
    )
    return r.json()

# Before any AI-assisted decision:
result = govern("user_8821", "loan_approval", risk_score=0.72)

if result["decision"] == "ALLOW":
    proceed_with_loan()
elif result["decision"] == "CHALLENGE":
    trigger_human_review()
else:
    reject_with_audit_trail(result["hash"])

JavaScript / Node 
async function govern(deviceId, action, riskScore = 0) {
  const res = await fetch('https://sebbi.pro/api/govern', {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${API_KEY}`,
      'Content-Type': 'application/json'
    },
    body: JSON.stringify({ device_id: deviceId, action, risk_score: riskScore })
  });
  return res.json();
}

// Usage — wraps any AI decision point
const { decision, hash } = await govern('user_8821', 'content_moderation', 0.6);
console.log(`Decision: ${decision} | Audit hash: ${hash}`);
